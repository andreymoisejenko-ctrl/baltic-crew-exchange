// A small, dependency-free ZIP writer (store method). MP4 is already compressed.
const enc=new TextEncoder();
const crcTable=new Uint32Array(256);for(let n=0;n<256;n++){let c=n;for(let k=0;k<8;k++)c=c&1?0xedb88320^(c>>>1):c>>>1;crcTable[n]=c>>>0}
function crc(bytes){let v=0xffffffff;for(let k=0;k<bytes.length;k++)v=crcTable[(v^bytes[k])&255]^(v>>>8);return(v^0xffffffff)>>>0}
function zipStore(files){let offset=0,parts=[],central=[];
 for(let f of files){const name=enc.encode(f.name);const bytes=f.data instanceof Uint8Array?f.data:enc.encode(f.data);let val=crc(bytes);
  let h=new Uint8Array(30+name.length),dv=new DataView(h.buffer);dv.setUint32(0,0x04034b50,true);dv.setUint16(4,20,true);dv.setUint16(6,0x0800,true);dv.setUint16(8,0,true);
  dv.setUint32(14,val,true);dv.setUint32(18,bytes.length,true);dv.setUint32(22,bytes.length,true);dv.setUint16(26,name.length,true);h.set(name,30);parts.push(h,bytes);
  let c=new Uint8Array(46+name.length),dc=new DataView(c.buffer);dc.setUint32(0,0x02014b50,true);dc.setUint16(4,20,true);dc.setUint16(6,20,true);dc.setUint16(8,0x0800,true);dc.setUint16(10,0,true);dc.setUint32(16,val,true);dc.setUint32(20,bytes.length,true);dc.setUint32(24,bytes.length,true);dc.setUint16(28,name.length,true);dc.setUint32(42,offset,true);c.set(name,46);central.push(c);offset+=h.length+bytes.length;
 }
 const size=central.reduce((n,v)=>n+v.length,0),end=new Uint8Array(22),de=new DataView(end.buffer);de.setUint32(0,0x06054b50,true);de.setUint16(8,files.length,true);de.setUint16(10,files.length,true);de.setUint32(12,size,true);de.setUint32(16,offset,true);
 return new Blob([...parts,...central,end],{type:'application/zip'});
}

<script id="kite-save-one-click-v64">
(function(){
 'use strict';
 if(document.documentElement.dataset.public==='true')return;
 const bridge=window.kiteSaveBridge;
 if(!bridge){console.error('Kite editor save bridge unavailable');return;}
 const {exportHTML,snapshot,store,saveFile}=bridge;
 const buttons=['quickSaveFile','quickSavePanel'].map(id=>document.getElementById(id));
 const toast=document.getElementById('quickSaveToast');
 let saveHandle=null, busy=false, toastTimer=null;
 const DB_NAME='kiteHTMLSavedFiles';
 const DB_STORE='files';
 const DB_KEY='kitesurfing-current-editor';
 function openHandleDB(){return new Promise((ok,fail)=>{
   if(!window.indexedDB){fail(new Error('IndexedDB unsupported'));return}
   const r=indexedDB.open(DB_NAME,1);
   r.onupgradeneeded=()=>{if(!r.result.objectStoreNames.contains(DB_STORE))r.result.createObjectStore(DB_STORE)};
   r.onsuccess=()=>ok(r.result);r.onerror=()=>fail(r.error);
 });}
 async function rememberHandle(handle){
   try{const d=await openHandleDB();await new Promise((ok,fail)=>{
      const t=d.transaction(DB_STORE,'readwrite');t.objectStore(DB_STORE).put(handle,DB_KEY);
      t.oncomplete=ok;t.onerror=()=>fail(t.error);
   });d.close();}catch(e){} 
 }
 async function restoreHandle(){
   try{const d=await openHandleDB();const h=await new Promise((ok,fail)=>{
       const q=d.transaction(DB_STORE,'readonly').objectStore(DB_STORE).get(DB_KEY);
       q.onsuccess=()=>ok(q.result);q.onerror=()=>fail(q.error);
   });d.close();if(h && h.kind==='file'){saveHandle=h;updateButtonLabel();}}
   catch(e){}
 }
 function showMessage(message,warning=false){
   toast.textContent=message;toast.style.borderColor=warning?'#ffb6a466':'#d8ff7366';
   toast.classList.add('show');clearTimeout(toastTimer);
   toastTimer=setTimeout(()=>toast.classList.remove('show'),4600);
   const st=document.getElementById('editorStatus');if(st)st.textContent=message;
 }
 function updateButtonLabel(){
   for(const b of buttons){if(!b)continue;
     b.textContent=saveHandle?'💾 Сохранить':'💾 Сохранить';
     b.title=saveHandle?('Сохранить изменения в файл '+saveHandle.name):'Сохранить редактируемый HTML на компьютер';
   }
 }
 // Called synchronously by an explicit click. The picker is opened BEFORE any
 // awaited operation so the browser does not lose transient user activation.
 async function savePresentation(){
   if(busy)return;
   busy=true;for(const b of buttons){if(b)b.disabled=true}
   let mode='download';
   try{
     let handle=saveHandle;
     if(!handle && typeof window.showSaveFilePicker==='function'){
       try{
         handle=await window.showSaveFilePicker({
            suggestedName:'Kaitserfings_mans_redaktors.html',
            types:[{description:'Редактируемая HTML-презентация',accept:{'text/html':['.html']}}]
         });
         saveHandle=handle;updateButtonLabel();
         // Do not wait for IndexedDB before writing; file permissions may be transient.
         rememberHandle(handle);
       }catch(err){
         if(err && err.name==='AbortError'){showMessage('Сохранение отменено: файл не выбран.',true);return}
         // Safari, in-app previews, policy-limited contexts: fall back to file download.
         handle=null;
       }
     }
     const html=exportHTML(false);
     if(handle){
       try{
         const perm=typeof handle.queryPermission==='function'?await handle.queryPermission({mode:'readwrite'}):'granted';
         if(perm!=='granted' && typeof handle.requestPermission==='function'){
           const result=await handle.requestPermission({mode:'readwrite'});
           if(result!=='granted'){showMessage('Доступ к файлу не разрешён. Выбери сохранение снова.',true);return}
         }
         const output=await handle.createWritable();
         await output.write(html);
         await output.close();
         mode='disk';
       }catch(e){
         // Do NOT claim disk save if permission/write failed.
         saveHandle=null;updateButtonLabel();
         showMessage('Не удалось перезаписать файл; скачиваю новую HTML-копию.',true);
       }
     }
     if(mode!=='disk'){
       const data=new Blob([html],{type:'text/html;charset=utf-8'});
       saveFile(data,'Kaitserfings_mans_redaktors.html');
       mode='download';
     }
     // The editor's existing draft store is a useful extra backup, not the only save.
     try{await store(snapshot())}catch(e){}
     if(mode==='disk')showMessage('✓ Сохранено в «'+handle.name+'» — фотографии и тексты внутри HTML.');
     else showMessage('✓ HTML скачан. Открой последний файл из папки «Загрузки» для продолжения.');
   }catch(e){showMessage('Не удалось сохранить: '+(e?.message||String(e)),true)}
   finally{busy=false;for(const b of buttons){if(b)b.disabled=false}}
 }
 for(const button of buttons){if(button)button.addEventListener('click',savePresentation)}
 window.addEventListener('keydown',function(e){
   if((e.ctrlKey||e.metaKey) && !e.shiftKey && e.key.toLowerCase()==='s'){
     e.preventDefault();e.stopImmediatePropagation();savePresentation();
   }
 },true);
 // File System Access handles are restored when the file remains under the same browser origin.
 restoreHandle();updateButtonLabel();
})();
</script>
